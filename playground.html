    Playwright 4-Step Lesson Example

    Copy this into a .spec.js file to test this playground end-to-end:

    import { test, expect } from '@playwright/test';

test('junior QA playground flow', async ({ page }) => {
  // 1. Navigate
  await page.goto('https://arjawad-qa.vercel.app/playground.html');

  // 2. Locate & 3. Act
  await page.getByLabel('Full name').fill('Junior QA');
  await page.getByRole('radio', { name: 'QA Sandbox' }).check();
  await page.getByRole('button', { name: 'Submit Form' }).click();

  // 4. Assert
  await expect(page.getByTestId('form-result')).toContainText('Junior QA');
  await expect(page.getByTestId('form-result')).toContainText('QA Sandbox');
});
